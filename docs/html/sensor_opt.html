27def compute_pod(snapshots, variance_target=0.90, min_modes=2):
28    """Compute a truncated POD basis from an ensemble of fields.
29
30    Parameters
31    ----------
32    snapshots : ndarray, shape (n_points, n_cases)
33        Field values, one column per case.
34    variance_target : float, optional
35        Fraction of the fluctuation energy (sum of squared singular values)
36        the retained modes must capture. Default 0.90.
37    min_modes : int, optional
38        Lower bound on the number of retained modes. Default 2.
39
40    Returns
41    -------
42    U_r : ndarray, shape (n_points, r)
43        Leading ``r`` left singular vectors of the mean-subtracted ensemble.
44    mean : ndarray, shape (n_points, 1)
45        Ensemble mean field.
46    r : int
47        Number of retained modes, ``max(min_modes, k)`` where ``k`` is the
48        smallest count reaching ``variance_target``.
49
50    Raises
51    ------
52    CfdDataError
53        If ``snapshots`` contains NaN or infinite values.
54    """
55    snapshots = np.asarray(snapshots)
56    _check_finite("snapshots", snapshots)
57
58    mean = snapshots.mean(axis=1, keepdims=True)
59    U, sigma, _ = np.linalg.svd(snapshots - mean, full_matrices=False)
60    cumulative = np.cumsum(sigma**2) / np.sum(sigma**2)
61    r = int(max(min_modes, np.searchsorted(cumulative, variance_target) + 1))
62    return U[:, :r], mean, r

def greedy_sensor_placement(U_r, n_sensors, eps=1e-10, coords=None, verbose=False, min_dist=0.0): 
 65def greedy_sensor_placement(U_r, n_sensors, eps=1e-10, coords=None, verbose=False, min_dist=0.0):
 66    """Select sensor locations by greedy D-optimal design.
 67
 68    At each step the point whose basis row ``w`` maximises the determinant
 69    gain ``1 + w (D^T D + eps I)^-1 w^T`` is added, and the inverse is
 70    updated with the Sherman-Morrison formula. With ``min_dist > 0``, every
 71    candidate closer than ``min_dist`` to a chosen sensor is removed after
 72    each pick.
 73
 74    Parameters
 75    ----------
 76    U_r : ndarray, shape (n_points, r)
 77        Basis restricted to the candidate points.
 78    n_sensors : int
 79        Number of sensors to place. Must not exceed ``n_points``.
 80    eps : float, optional
 81        Regularisation of the initial information matrix. Default 1e-10.
 82    coords : ndarray, shape (n_points, 3), optional
 83        Candidate coordinates in metres. Required when ``min_dist > 0``;
 84        otherwise only used in progress messages.
 85    verbose : bool, optional
 86        Print each selected sensor. Default False.
 87    min_dist : float, optional
 88        Minimum distance between sensors in metres. Default 0 (no spacing
 89        constraint; identical picks to the unconstrained algorithm).
 90
 91    Returns
 92    -------
 93    list of int
 94        Row indices into ``U_r`` in selection order; all distinct.
 95
 96    Raises
 97    ------
 98    CfdDataError
 99        If ``U_r`` contains NaN or infinite values.
100    ValueError
101        If ``n_sensors`` exceeds the number of candidate points, if
102        ``min_dist > 0`` without ``coords``, or if the spacing constraint
103        leaves too few candidates.
104    """
105    U_r = np.asarray(U_r)
106    _check_finite("U_r", U_r)
107    n_points, r = U_r.shape
108    if n_sensors > n_points:
109        raise ValueError(f"Cannot place {n_sensors} sensors on {n_points} candidate points.")
110    if min_dist > 0:
111        if coords is None:
112            raise ValueError("min_dist > 0 needs coords.")
113        coords = np.asarray(coords, dtype=float)
114
115    selected = []
116    available = np.ones(n_points, dtype=bool)
117
118    DtD_inv = (1/eps) * np.eye(r)
119
120    for k in range(n_sensors):
121        if not available.any():
122            raise ValueError(f"Only {k} sensors fit with min_dist={min_dist} m; asked for {n_sensors}.")
123        gains = 1.0 + ((U_r @ DtD_inv) * U_r).sum(axis=1)
124        gains[~available] = -np.inf
125        best_idx = int(np.argmax(gains))
126        best_gain = gains[best_idx]
127
128        selected.append(best_idx)
129        available[best_idx] = False
130        if min_dist > 0:
131            available &= np.linalg.norm(coords - coords[best_idx], axis=1) >= min_dist
132
133        w = U_r[best_idx, :].reshape(-1, 1)
134        Dw = DtD_inv @ w
135        DtD_inv -= (Dw @ Dw.T) / (1.0 + (w.T @ Dw).item())
136
137        if verbose:
138            coord_str = f"coords {coords[best_idx]}" if coords is not None else ""
139            print(f"Sensor {k+1}: local candidate index {best_idx}, {coord_str}, gain {best_gain:.4f}")
140
141    return selected

144def reconstruct(U_r, X_mean, selected_indices, y_measured):
145    """Reconstruct a full field from sparse sensor readings.
146
147    Solves ``min_z || U_r[selected] z - (y - mean[selected]) ||`` by least
148    squares and returns ``U_r z + mean``.
149
150    Parameters
151    ----------
152    U_r : ndarray, shape (n_points, r)
153        POD basis over the full mesh.
154    X_mean : ndarray, shape (n_points, 1)
155        Mean field the basis was built around.
156    selected_indices : array_like of int
157        Mesh indices of the sensors.
158    y_measured : ndarray, shape (n_sensors,)
159        Readings at ``selected_indices``.
160
161    Returns
162    -------
163    ndarray, shape (n_points,)
164        Reconstructed field.
165
166    Raises
167    ------
168    CfdDataError
169        If ``y_measured`` contains NaN or infinite values.
170    """
171    _check_finite("y_measured", y_measured)
172    D = U_r[selected_indices, :]
173    y = y_measured - X_mean[selected_indices, 0]
174    z, _, _, _ = np.linalg.lstsq(D, y, rcond=None)
175    Cp_reconstructed = U_r @ z + X_mean[:, 0]
176    return Cp_reconstructed
